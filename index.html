Swift Fundamentals for Interviews

The language quirks interviewers expect you to reason about, beyond generic CS knowledge.

Value vs. reference types

struct/enum are value types (copied on assignment, copy-on-write for collections); class is a reference type (shared, ARC-managed). Interview implication: a linked list node or tree node is almost always a class, so multiple references can point at the same node.

Optionals

func firstEven(_ nums: [Int]) -> Int? {
    for n in nums where n % 2 == 0 { return n }
    return nil
}

if let value = firstEven([1, 3, 4, 5]) {
    print(value)          // 4
}

let value = firstEven([1, 3, 5]) ?? -1   // nil-coalescing

Prefer guard let for early exits and avoid force-unwrapping (!) unless you've just proven the value exists.

    String(s.reversed())
}

// Two arrays of characters, without index arithmetic on String
func isAnagram(_ a: String, _ b: String) -> Bool {
    a.count == b.count && a.sorted() == b.sorted()
}
Strings are collections of Character (extended grapheme clusters), not bytes — s[i] integer subscripting doesn't exist. Convert to Array(s) first if you need repeated random access.

Stack & Queue DATA STRUCTURE

struct Stack<T> {
    private var items: [T] = []
    var isEmpty: Bool { items.isEmpty }
    mutating func push(_ item: T) { items.append(item) }
    mutating func pop() -> T? { items.popLast() }
    func peek() -> T? { items.last }
}

struct Queue<T> {
    private var items: [T] = []
    mutating func enqueue(_ item: T) { items.append(item) }
    mutating func dequeue() -> T? {
        items.isEmpty ? nil : items.removeFirst()   // O(n); use two stacks or a deque for O(1)
    }
}
removeFirst() on Array is O(n). For an O(1) queue, use two stacks (amortized O(1)) or Swift Collections' Deque.

Linked List DATA STRUCTURE

final class ListNode {
    var val: Int
    var next: ListNode?
    init(_ val: Int, _ next: ListNode? = nil) {
        self.val = val
        self.next = next
    }
}

func reverseList(_ head: ListNode?) -> ListNode? {
    var prev: ListNode? = nil
    var current = head
    while let node = current {
        let next = node.next
        node.next = prev
        prev = node
        current = next
    }
    return prev
}

// Floyd's cycle detection
func hasCycle(_ head: ListNode?) -> Bool {
    var slow = head, fast = head
    while fast != nil && fast?.next != nil {
        slow = slow?.next
        fast = fast?.next?.next
        if slow === fast { return true }
    }
    return false
}

Trees & Binary Search Trees DATA STRUCTURE

final class TreeNode {
    var val: Int
    var left: TreeNode?
    var right: TreeNode?
    init(_ val: Int) { self.val = val }
}

func inorder(_ node: TreeNode?, _ result: inout [Int]) {
    guard let node else { return }
    inorder(node.left, &result)
    result.append(node.val)
    inorder(node.right, &result)
}

func insert(_ root: TreeNode?, _ val: Int) -> TreeNode {
    guard let root else { return TreeNode(val) }
    if val < root.val { root.left = insert(root.left, val) }
    else { root.right = insert(root.right, val) }
    return root
}

func isBalanced(_ root: TreeNode?) -> Bool {
    func height(_ node: TreeNode?) -> Int {
        guard let node else { return 0 }
        let l = height(node.left), r = height(node.right)
        if l == -1 || r == -1 || abs(l - r) > 1 { return -1 }
        return max(l, r) + 1
    }
    return height(root) != -1
}

Trie DATA STRUCTURE

final class TrieNode {
    var children: [Character: TrieNode] = [:]
    var isWord = false
}

final class Trie {
    private let root = TrieNode()

    func insert(_ word: String) {
        var node = root
        for char in word {
            node = node.children[char] ?? {
                let newNode = TrieNode()
                node.children[char] = newNode
                return newNode
            }()
        }
        node.isWord = true
    }

    func search(_ word: String) -> Bool {
        node(for: word)?.isWord ?? false
    }

    func startsWith(_ prefix: String) -> Bool {
        node(for: prefix) != nil
    }

    private func node(for word: String) -> TrieNode? {
        var node = root
        for char in word {
            guard let next = node.children[char] else { return nil }
            node = next
        }
        return node
    }
}

Heap / Priority Queue DATA STRUCTURE

The standard library has no built-in heap. Model one with an array and sift operations.

struct MinHeap<T: Comparable> {
    private var items: [T] = []
    var isEmpty: Bool { items.isEmpty }
    var count: Int { items.count }
    var peek: T? { items.first }

    mutating func insert(_ value: T) {
        items.append(value)
        siftUp(from: items.count - 1)
    }

    mutating func extractMin() -> T? {
        guard !items.isEmpty else { return nil }
        items.swapAt(0, items.count - 1)
        let min = items.removeLast()
        siftDown(from: 0)
        return min
    }

    private mutating func siftUp(from index: Int) {
        var child = index
        while child > 0 {
            let parent = (child - 1) / 2
            if items[child] < items[parent] {
                items.swapAt(child, parent)
                child = parent
            } else { break }
        }
    }

    private mutating func siftDown(from index: Int) {
        var parent = index
        while true {
            let left = 2 * parent + 1, right = 2 * parent + 2
            var smallest = parent
            if left < items.count && items[left] < items[smallest] { smallest = left }
            if right < items.count && items[right] < items[smallest] { smallest = right }
            if smallest == parent { break }
            items.swapAt(parent, smallest)
            parent = smallest
        }
    }
}
For contest-speed code, Swift's Array combined with sorting is often "good enough," but a real heap is the expected answer for "kth largest," "merge k lists," or "top-k" problems.

Graph (BFS / DFS) DATA STRUCTURE

// Adjacency list
var graph: [Int: [Int]] = [
    0: [1, 2],
    1: [0, 3],
    2: [0, 3],
    3: [1, 2]
]

func bfs(_ start: Int, _ graph: [Int: [Int]]) -> [Int] {
    var visited: Set<Int> = [start]
    var queue = [start]
    var order: [Int] = []
    var head = 0
    while head < queue.count {
        let node = queue[head]; head += 1
        order.append(node)
        for neighbor in graph[node] ?? [] where !visited.contains(neighbor) {
            visited.insert(neighbor)
            queue.append(neighbor)
        }
    }
    return order
}

func dfs(_ node: Int, _ graph: [Int: [Int]], _ visited: inout Set<Int>, _ order: inout [Int]) {
    guard !visited.contains(node) else { return }
    visited.insert(node)
    order.append(node)
    for neighbor in graph[node] ?? [] {
        dfs(neighbor, graph, &visited, &order)
    }
}

Binary Search

func binarySearch(_ arr: [Int], _ target: Int) -> Int? {
    var low = 0, high = arr.count - 1
    while low <= high {
        let mid = low + (high - low) / 2   // avoids overflow
        if arr[mid] == target { return mid }
        if arr[mid] < target { low = mid + 1 }
        else { high = mid - 1 }
    }
    return nil
}

// Search in rotated sorted array
func searchRotated(_ nums: [Int], _ target: Int) -> Int? {
    var low = 0, high = nums.count - 1
    while low <= high {
        let mid = (low + high) / 2
        if nums[mid] == target { return mid }
        if nums[low] <= nums[mid] {
            if nums[low] <= target && target < nums[mid] { high = mid - 1 }
            else { low = mid + 1 }
        } else {
            if nums[mid] < target && target <= nums[high] { low = mid + 1 }
            else { high = mid - 1 }
        }
    }
    return nil
}

Two Pointers & Sliding Window

// Two Sum on a sorted array — two pointers
func twoSumSorted(_ nums: [Int], _ target: Int) -> (Int, Int)? {
    var left = 0, right = nums.count - 1
    while left < right {
        let sum = nums[left] + nums[right]
        if sum == target { return (left, right) }
        if sum < target { left += 1 } else { right -= 1 }
    }
    return nil
}

// Longest substring without repeating characters — sliding window
func lengthOfLongestSubstring(_ s: String) -> Int {
    var seen: Set<Character> = []
    var left = s.startIndex
    var best = 0
    var right = s.startIndex
    while right < s.endIndex {
        let char = s[right]
        while seen.contains(char) {
            seen.remove(s[left])
            left = s.index(after: left)
        }
        seen.insert(char)
        best = max(best, s.distance(from: left, to: right) + 1)
        right = s.index(after: right)
    }
    return best
}

Dynamic Programming

// Climbing Stairs — bottom-up tabulation
func climbStairs(_ n: Int) -> Int {
    guard n > 2 else { return n }
    var prev = 1, curr = 2
    for _ in 3...n {
        (prev, curr) = (curr, prev + curr)
    }
    return curr
}

// Coin Change — minimum coins to make an amount
func coinChange(_ coins: [Int], _ amount: Int) -> Int {
    var dp = Array(repeating: amount + 1, count: amount + 1)
    dp[0] = 0
    for total in 1...amount {
        for coin in coins where coin <= total {
            dp[total] = min(dp[total], dp[total - coin] + 1)
        }
    }
    return dp[amount] > amount ? -1 : dp[amount]
}

// Longest Common Subsequence — memoized top-down
func longestCommonSubsequence(_ a: String, _ b: String) -> Int {
    let a = Array(a), b = Array(b)
    var memo = [[Int: Int]](repeating: [:], count: a.count + 1)

    func lcs(_ i: Int, _ j: Int) -> Int {
        if i == a.count || j == b.count { return 0 }
        if let cached = memo[i][j] { return cached }
        let result: Int
        if a[i] == b[j] {
            result = 1 + lcs(i + 1, j + 1)
        } else {
            result = max(lcs(i + 1, j), lcs(i, j + 1))
        }
        memo[i][j] = result
        return result
    }
    return lcs(0, 0)
}

Valid Parentheses

func isValid(_ s: String) -> Bool {
    let pairs: [Character: Character] = [")": "(", "]": "[", "}": "{"]
    var stack: [Character] = []
    for char in s {
        if let opener = pairs[char] {
            if stack.popLast() != opener { return false }
        } else {
            stack.append(char)
        }
    }
    return stack.isEmpty
}

func merge(_ intervals: [[Int]]) -> [[Int]] {
    let sorted = intervals.sorted { $0[0] < $1[0] }

        } else {
            result.append(interval)
        }

LRU Cache

final class LRUCache {
    private final class Node {
        let key: Int
        var value: Int
        var prev: Node?
        var next: Node?
        init(_ key: Int, _ value: Int) { self.key = key; self.value = value }
    }

    private let capacity: Int
    private var map: [Int: Node] = [:]
    private let head = Node(-1, -1)
    private let tail = Node(-1, -1)

    init(_ capacity: Int) {
        self.capacity = capacity
        head.next = tail
        tail.prev = head
    }

    private func remove(_ node: Node) {
        node.prev?.next = node.next
        node.next?.prev = node.prev
    }

    private func insertFront(_ node: Node) {
        node.next = head.next
        node.prev = head
        head.next?.prev = node
        head.next = node
    }

    func get(_ key: Int) -> Int {
        guard let node = map[key] else { return -1 }
        remove(node)
        insertFront(node)
        return node.value
    }

    func put(_ key: Int, _ value: Int) {
        if let existing = map[key] { remove(existing) }
        let node = Node(key, value)
        map[key] = node
        insertFront(node)
        if map.count > capacity, let lru = tail.prev, lru !== head {
            remove(lru)
            map.removeValue(forKey: lru.key)
        }
    }
}

Kth Largest Element

// O(n log k) using a fixed-size min-heap: keep only the k largest seen so far.
func findKthLargest(_ nums: [Int], _ k: Int) -> Int {
    var heap = MinHeap<Int>()
    for num in nums {
        heap.insert(num)
        if heap.count > k { _ = heap.extractMin() }
    }
    return heap.peek ?? -1
}

// Simpler to reach for under time pressure: sort and index, O(n log n).
func findKthLargestSimple(_ nums: [Int], _ k: Int) -> Int {
    nums.sorted(by: >)[k - 1]
}

Swift-Specific Gotchas

  • String indexing isn't O(1). s[s.index(s.startIndex, offsetBy: i)] walks the string. Convert to Array(s) once if you need repeated random access.
  • Copy-on-write. Assigning an Array/Dictionary to a new variable doesn't copy memory until one side mutates — useful to know when reasoning about space complexity out loud.
  • Class vs struct for nodes. Linked lists, trees, and graphs need reference semantics so multiple pointers can alias the same node — always model nodes as class, not struct.
  • Overflow. Swift traps on integer overflow by default (Int addition, etc.). Use low + (high - low) / 2 for binary search midpoints, not (low + high) / 2.
  • Optionals over sentinels. Return nil instead of -1 where possible — it documents "no answer" in the type system and interviewers notice.
  • Identity vs. equality. Use === to compare object identity (e.g., detecting a cycle by reference), and == for value equality.
  • No built-in heap/deque. The standard library ships Array, Dictionary, Set — be ready to hand-roll a heap (shown above) since interviewers know Swift lacks one out of the box.
